## 1. 로그인 폼

### 출처: `sp16 LoginForm.jsp`

### 클라이언트 (loginForm.jsp)
```jsp
<%@ page contentType="text/html; charset=UTF-8"%>
<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>로그인</title>
<script type="text/javascript" src="http://code.jquery.com/jquery.min.js"></script>
<script type="text/javascript">

$(function(){
    $("#submitBtn").click(function(){
        $("#error").css("display", "none");
        
        if ($("#loginId").val() == "" || $("#password").val() == "") {
            $("#error").html("아이디와 비밀번호를 모두 입력해주세요")
                       .css("display", "inline");
            return;
        }
        $("#loginForm").submit();
    });
});

</script>
</head>
<body>

<h1>로그인</h1>

<form action="${pageContext.request.contextPath}/user/login" method="post" id="loginForm">
    <table>
        <tr>
            <th><label for="loginId">아이디</label></th>
            <td><input type="text" id="loginId" name="loginId" placeholder="아이디"></td>
        </tr>
        <tr>
            <th><label for="password">비밀번호</label></th>
            <td><input type="password" id="password" name="password" placeholder="비밀번호"></td>
        </tr>
        <tr>
            <td colspan="2" align="center">
                <button type="button" id="submitBtn" class="btn">로그인</button>
                <button type="reset" class="btn">취소</button>
                <br><br>
                <span id="error" style="color: red; font-size: small; display: none;"></span>
            </td>
        </tr>
    </table>
</form>

</body>
</html>
```

### 서버 (UserController.java)
```java
@PostMapping("/login")
public String login(String loginId, String password, HttpSession session, Model model) {
    User user = service.login(loginId, password);
    
    if (user != null) {
        session.setAttribute("loginUser", user);
        return "redirect:/main";
    } else {
        model.addAttribute("errMsg", "아이디 또는 비밀번호가 틀렸습니다.");
        return "user/loginForm";
    }
}

@GetMapping("/logout")
public String logout(HttpSession session) {
    session.invalidate();
    return "redirect:/user/login";
}
```

## 2. 등록 폼

### 출처: `sp16 write.jsp` (게시판)

### 클라이언트 (write.jsp 변형 — 카페 등록 예)
```jsp
<%@ page contentType="text/html; charset=UTF-8"%>
<%@ taglib prefix="c" uri="http://java.sun.com/jsp/jstl/core"%>
<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>카페 등록</title>
<link rel="stylesheet" type="text/css" 
      href="${pageContext.request.contextPath}/resources/css/writeStyle.css">
<script type="text/javascript" src="http://code.jquery.com/jquery.min.js"></script>
<script type="text/javascript">

function sendOk() {
    $("#error").css("display", "none");
    
    const f = document.cafeForm;
    
    let str = f.cafeName.value.trim();
    if (!str) {
        $("#error").html("카페명을 입력해주세요").css("display", "inline");
        f.cafeName.focus();
        return;
    }
    
    str = f.businessNumber.value.trim();
    if (!str) {
        $("#error").html("사업자번호를 입력해주세요").css("display", "inline");
        f.businessNumber.focus();
        return;
    }
    
    str = f.address.value.trim();
    if (!str) {
        $("#error").html("주소를 입력해주세요").css("display", "inline");
        f.address.focus();
        return;
    }
    
    f.action = "${pageContext.request.contextPath}/cafe/${mode}";
    f.submit();
}

</script>
</head>
<body>

<div class="body-container">
    <div class="body-title">
        <h3>카페 등록</h3>
    </div>
    
    <form name="cafeForm" method="post">
        <table class="table table-border table-form">
            <tr>
                <td>카페명</td>
                <td><input type="text" name="cafeName" maxlength="100" 
                           class="form-control" value="${dto.cafeName}"></td>
            </tr>
            <tr>
                <td>대표자명</td>
                <td><input type="text" name="ownerName" maxlength="50" 
                           class="form-control" value="${dto.ownerName}"></td>
            </tr>
            <tr>
                <td>사업자번호</td>
                <td><input type="text" name="businessNumber" maxlength="10" 
                           class="form-control" value="${dto.businessNumber}"></td>
            </tr>
            <tr>
                <td>주소</td>
                <td><input type="text" name="address" 
                           class="form-control" value="${dto.address}"></td>
            </tr>
            <tr>
                <td>전화번호</td>
                <td><input type="text" name="phone" maxlength="11" 
                           class="form-control" value="${dto.phone}"></td>
            </tr>
            <tr>
                <td>카페소개</td>
                <td><textarea name="description" maxlength="4000" 
                              class="form-control">${dto.description}</textarea></td>
            </tr>
        </table>
        
        <table class="table">
            <tr>
                <td align="center">
                    <button type="button" class="btn" onclick="sendOk();">
                        ${mode=='update' ? '수정완료' : "등록완료"}
                    </button>
                    <button type="reset" class="btn">다시입력</button>
                    <button type="button" class="btn" 
                            onclick="location.href='${pageContext.request.contextPath}/cafe/list';">
                        ${mode=='update' ? '수정취소' : "등록취소"}
                    </button>
                    <c:if test="${mode=='update'}">
                        <input type="hidden" name="cafeId" value="${dto.cafeId}">
                    </c:if>
                    <br><br>
                    <span id="error" style="color: red; font-size: small; display: none;"></span>
                </td>
            </tr>
        </table>
    </form>
</div>

</body>
</html>
```

### 핵심 포인트
- `${mode=='update' ? '수정완료' : "등록완료"}` — 등록/수정 폼 같이 쓰기
- `f.action = "${pageContext.request.contextPath}/cafe/${mode}";` — JS에서 동적으로 action 설정
- `<c:if test="${mode=='update'}">` — 수정 모드일 때만 hidden input 추가

## 3. 목록 + 페이징 + 검색

### 출처: `sp16 list.jsp`

### 클라이언트 (list.jsp)
```jsp
<%@ page contentType="text/html; charset=UTF-8"%>
<%@ taglib prefix="c" uri="http://java.sun.com/jsp/jstl/core"%>
<%@ taglib prefix="fmt" uri="http://java.sun.com/jsp/jstl/fmt"%>
<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>카페 목록</title>
<link rel="icon" href="data:;base64,iVBORw0KGgo=">
<link rel="stylesheet" type="text/css" 
      href="${pageContext.request.contextPath}/resources/css/listStyle.css">
<script type="text/javascript">

// 엔터 키로 검색
document.addEventListener("DOMContentLoaded", () => {
    const inputEL = document.querySelector("form input[name=kwd]");
    inputEL.addEventListener("keydown", function(evt) {
        if (evt.key == "Enter") {
            evt.preventDefault();
            searchList();
        }
    });
});

// 검색 함수
function searchList() {
    const f = document.searchForm;
    
    if (!f.kwd.value.trim()) {
        return;    // 빈 검색어 무시
    }
    
    const formData = new FormData(f);
    let requestParam = new URLSearchParams(formData).toString();
    
    let url = "${pageContext.request.contextPath}/cafe/list";
    location.href = url + "?" + requestParam;
}

</script>
</head>
<body>

<div class="body-container">
    <div class="body-title">
        <h3><span>|</span>카페 목록</h3>
    </div>
    
    <table class="table">
        <tr>
            <td width="50%">/ ${dataCount}개</td>
            <td align="right">&nbsp;</td>
        </tr>
    </table>
    
    <table class="table table-border table-list">
        <thead>
            <tr>
                <td class="num">번호</td>
                <td class="subject">카페명</td>
                <td class="name">대표자</td>
                <td class="date">등록일</td>
                <td class="hit">조회수</td>
            </tr>
        </thead>
        <tbody>
            <c:forEach var="dto" items="${list}">
                <tr>
                    <td>${dto.cafeId}</td>
                    <td class="left">
                        <a href="${pageContext.request.contextPath}/cafe/article?cafeId=${dto.cafeId}&${query}">
                            ${dto.cafeName}
                        </a>
                    </td>
                    <td>${dto.ownerName}</td>
                    <td>${dto.regDate}</td>
                    <td>${dto.hitCount}</td>
                </tr>
            </c:forEach>
        </tbody>
    </table>
    
    <div class="page-navigation">
        ${dataCount == 0 ? "등록된 카페가 없습니다" : paging}
    </div>
    
    <table class="table">
        <tr>
            <td width="100">
                <button type="button" class="btn" 
                        onclick="location.href='${pageContext.request.contextPath}/cafe/list';">
                    새로고침
                </button>
            </td>
            <td align="center">
                <form name="searchForm">
                    <select name="schType" class="form-select">
                        <option value="all" ${schType=="all" ? "selected" : ""}>전체</option>
                        <option value="cafeName" ${schType=="cafeName" ? "selected" : ""}>카페명</option>
                        <option value="ownerName" ${schType=="ownerName" ? "selected" : ""}>대표자</option>
                    </select>
                    <input type="text" name="kwd" class="form-control" value="${kwd}">
                    <button type="button" class="btn1" onclick="searchList()">검색</button>
                </form>
            </td>
            <td align="right" width="100">
                <button type="button" class="btn" 
                        onclick="location.href='${pageContext.request.contextPath}/cafe/enroll';">
                    카페등록
                </button>
            </td>
        </tr>
    </table>
</div>

</body>
</html>
```

### 핵심 포인트
- **엔터 키 검색**: `addEventListener("keydown")` + `evt.preventDefault()`
- **`new FormData(f)` + `URLSearchParams`** — 폼 전체를 한번에 쿼리스트링으로
- **선택 유지**: `${schType=="all" ? "selected" : ""}` (이전 선택 값 유지)
- **`${query}`** — 검색 조건 유지하면서 페이지 이동용

## 4. 상세 + 삭제확인

### 출처: `sp16 article.jsp`

### 클라이언트 (article.jsp)
```jsp
<%@ page contentType="text/html; charset=UTF-8"%>
<%@ taglib prefix="c" uri="http://java.sun.com/jsp/jstl/core"%>
<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>카페 상세</title>
<link rel="stylesheet" type="text/css" 
      href="${pageContext.request.contextPath}/resources/css/articleStyle.css">
<script type="text/javascript">

function deleteOk(cafeId) {
    if (confirm(cafeId + " 카페를 정말 삭제하시겠습니까?")) {
        let url = "${pageContext.request.contextPath}/cafe/delete?cafeId=" + cafeId + "&${query}";
        location.href = url;
    }
}

</script>
</head>
<body>

<div class="body-container">
    <div class="body-title">
        <h3>카페 상세</h3>
    </div>
    
    <table class="table table-border table-article">
        <thead>
            <tr>
                <td colspan="2" align="center">
                    <c:out value="${dto.cafeName}"/>
                </td>
            </tr>
        </thead>
        <tbody>
            <tr>
                <td width="50%">대표자: ${dto.ownerName}</td>
                <td align="right">${dto.regDate} | 조회 ${dto.hitCount}</td>
            </tr>
            <tr>
                <td colspan="2" valign="top" height="200">
                    ${dto.description}
                </td>
            </tr>
            <tr>
                <td colspan="2">
                    이전: 
                    <c:if test="${not empty prevDto}">
                        <a href="${pageContext.request.contextPath}/cafe/article?cafeId=${prevDto.cafeId}&${query}">
                            <c:out value="${prevDto.cafeName}"/>
                        </a>
                    </c:if>
                </td>
            </tr>
            <tr>
                <td colspan="2">
                    다음: 
                    <c:if test="${not empty nextDto}">
                        <a href="${pageContext.request.contextPath}/cafe/article?cafeId=${nextDto.cafeId}&${query}">
                            <c:out value="${nextDto.cafeName}"/>
                        </a>
                    </c:if>
                </td>
            </tr>
        </tbody>
    </table>
    
    <table class="table">
        <tr>
            <td width="50%">
                <button type="button" class="btn" 
                        onclick="location.href='${pageContext.request.contextPath}/cafe/update?cafeId=${dto.cafeId}&page=${page}'">
                    수정
                </button>
                <button type="button" class="btn" onclick="deleteOk('${dto.cafeId}')">
                    삭제
                </button>
            </td>
            <td align="right">
                <button type="button" class="btn" 
                        onclick="location.href='${pageContext.request.contextPath}/cafe/list?page=${page}';">
                    리스트
                </button>
            </td>
        </tr>
    </table>
</div>

</body>
</html>
```

### 핵심 포인트
- **`confirm()` 으로 삭제 재확인** — alert 안 쓰는 패턴 중 예외 (사용자 의사 확인 필요)
- **`<c:if test="${not empty prevDto}">`** — 이전/다음 글이 있을 때만 링크 보이기
- **`<c:out value="...">`** — XSS 방어 (특수문자 이스케이프)

### 서버 — JavaMailSender 사용 (간단 예시)
```java
@Autowired
private JavaMailSender mailSender;

@PostMapping("/mail/send")
public String sendMail(String receiveEmail, String subject, String content) throws Exception {
    SimpleMailMessage msg = new SimpleMailMessage();
    msg.setTo(receiveEmail);
    msg.setSubject(subject);
    msg.setText(content);
    mailSender.send(msg);
    return "mail/sendOk";
}
```

### application.yml에 메일 설정 추가

## 7. 회원가입 (풀스택)

### 출처: 강사님 패턴 통합 + noExit 적용

### enrollForm.jsp (전체)
```jsp
<%@ page contentType="text/html; charset=UTF-8"%>
<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>회원가입</title>
<script type="text/javascript" src="http://code.jquery.com/jquery.min.js"></script>
<script type="text/javascript">

$(function(){
    let isIdChecked = false;
    
    $("#idChkBtn").click(function(){
        $("#error").css("display", "none");
        
        let loginId = $("#loginId").val().trim();
        if (loginId === "") {
            $("#error").html("아이디를 입력해주세요.").css({color: "red", display: "inline"});
            $("#loginId").focus();
            return;
        }
        
        $.ajax({
            type: "POST",
            url: "${pageContext.request.contextPath}/user/id-check",
            data: { loginId: loginId },
            success: function(result){
                if (result === "AVAILABLE") {
                    $("#error").html("사용 가능한 아이디입니다.").css({color: "blue", display: "inline"});
                    $("#loginId").prop("readonly", true);
                    isIdChecked = true;
                } else {
                    $("#error").html("이미 사용 중인 아이디입니다.").css({color: "red", display: "inline"});
                    $("#loginId").val("").focus();
                }
            },
```

### UserController.java (전체)
```java
package com.noexit.app.controller;

import org.springframework.stereotype.Controller;
import org.springframework.web.bind.annotation.*;
import com.noexit.app.model.User;
import com.noexit.app.service.UserService;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;

@Controller
@Slf4j
@RequiredArgsConstructor
@RequestMapping("/user")
public class UserController {

    private final UserService service;

    @GetMapping("/enroll")
    public String enrollForm() { return "user/enrollForm"; }

    @PostMapping("/enroll")
    public String enroll(User user) {
        try { service.enroll(user); } 
        catch (Exception e) { log.info("enroll : ", e); }
        return "redirect:/user/login";
    }

    @PostMapping("/id-check")
    @ResponseBody
    public String idCheck(String loginId) {
        int count = service.countByLoginId(loginId);
        return count == 0 ? "AVAILABLE" : "DUPLICATED";
    }

    @GetMapping("/login")
    public String loginForm() { return "user/loginForm"; }

    @PostMapping("/login")
    public String login() { return "redirect:/main"; }

    @GetMapping("/logout")
    public String logout() { return "redirect:/user/login"; }
}
```

```java
public interface UserService {

    public int countByLoginId(String loginId);

@Service
@RequiredArgsConstructor
public class UserServiceImpl implements UserService {
    private final UserMapper userMapper;

    @Override
    @Transactional
    public void enroll(User user) {
        userMapper.insertAccount(user);
        userMapper.insertInfo(user);
    }

    @Override
    public int countByLoginId(String loginId) {
        return userMapper.countByLoginId(loginId);
    }
}
```
